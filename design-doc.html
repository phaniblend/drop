4. Core Modules Implementation

Module A: Playwright Supplier Scraper (scraper.ts)

Extracts pricing, variant definitions, and uncompressed media by reading window.runParams rather than unstable front-end DOM classes.

import { chromium } from 'playwright-extra';
import stealthPlugin from 'puppeteer-extra-plugin-stealth';

chromium.use(stealthPlugin());

export interface ParsedSupplierPayload {
  title: string;
  baseCost: number;
  galleryImages: string[];
  variants: Array<{
    skuId: string;
    attributes: string;
    cost: number;
    stock: number;
    imageUrl?: string;
  }>;
}

export async function scrapeSupplierUrl(targetUrl: string): Promise<ParsedSupplierPayload> {
  const browser = await chromium.launch({
    headless: true,
    args: ['--no-sandbox', '--disable-setuid-sandbox']
  });

  const context = await browser.newContext({
    userAgent: 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/126.0.0.0 Safari/537.36',
    viewport: { width: 1920, height: 1080 }
  });

  const page = await context.newPage();

  try {
    await page.goto(targetUrl, { waitUntil: 'domcontentloaded', timeout: 35000 });

    const rawPayload = await page.evaluate(() => {
      const scripts = Array.from(document.querySelectorAll('script'));
      const scriptWithParams = scripts.find(s => s.innerText && s.innerText.includes('window.runParams'));
      return scriptWithParams ? scriptWithParams.innerText : null;
    });

    await browser.close();

    if (!rawPayload) {
      throw new Error('Hydration script (window.runParams) not found on supplier page.');
    }

    return parseHydrationData(rawPayload);
  } catch (error) {
    await browser.close();
    throw error;
  }
}

function parseHydrationData(scriptContent: string): ParsedSupplierPayload {
  const matched = scriptContent.match(/data:\s*(\{.*?\})\s*,\s*csrfToken/s);
  if (!matched || !matched[1]) {
    throw new Error('Regex failed to isolate JSON data chunk from script');
  }

  const payload = JSON.parse(matched[1]);
  const productInfo = payload.productInfoComponent || {};
  const priceInfo = payload.priceComponent || {};
  const skuInfo = payload.skuComponent || {};
  const imageInfo = payload.imageComponent || {};

  return {
    title: productInfo.subject || 'Imported Wholesale Product',
    baseCost: parseFloat(priceInfo.origPrice?.minAmount?.value || '0.00'),
    galleryImages: (imageInfo.imagePathList || []).map((img: string) => img.replace(/_\d+x\d+\.jpg$/, '')),
    variants: (skuInfo.skuPriceList || []).map((sku: any) => ({
      skuId: String(sku.skuId),
      attributes: sku.skuAttr || 'Default',
      cost: parseFloat(sku.skuVal?.actSkuCalPrice || sku.skuVal?.skuCalPrice || '0.00'),
      stock: sku.skuVal?.availQuantity || 0,
      imageUrl: sku.skuPropertyImagePath
    }))
  };
}

Module B: Shopify GraphQL Publisher (publisher.ts)

Constructs and pushes products, variants, and markup pricing using the Shopify GraphQL Admin API (version 2026-07).

import { GraphQLClient, gql } from 'graphql-request';

const shopifyClient = new GraphQLClient(
  `https://${process.env.SHOPIFY_STORE_DOMAIN}.myshopify.com/admin/api/2026-07/graphql.json`,
  {
    headers: {
      'X-Shopify-Access-Token': process.env.SHOPIFY_ADMIN_TOKEN!,
      'Content-Type': 'application/json',
    },
  }
);

interface TransformedProductInput {
  title: string;
  descriptionHtml: string;
  tags: string[];
  variants: Array<{
    sku: string;
    cost: number;
    title: string;
  }>;
}

export async function publishProductToShopify(data: TransformedProductInput) {
  const mutation = gql`
    mutation ProductCreateWithVariants($input: ProductInput!) {
      productCreate(input: $input) {
        product {
          id
          title
          handle
          variants(first: 50) {
            edges {
              node {
                id
                sku
                price
              }
            }
          }
        }
        userErrors {
          field
          message
        }
      }
    }
  `;

  const variables = {
    input: {
      title: data.title,
      descriptionHtml: data.descriptionHtml,
      vendor: 'DropshipOS Node',
      status: 'DRAFT',
      tags: data.tags,
      variants: data.variants.map((v) => ({
        sku: v.sku,
        price: (v.cost * 3.0).toFixed(2),
        compareAtPrice: (v.cost * 4.5).toFixed(2),
        inventoryManagement: 'SHOPIFY'
      }))
    }
  };

  const res: any = await shopifyClient.request(mutation, variables);

  if (res.productCreate.userErrors.length > 0) {
    throw new Error(`Shopify publishing failed: ${JSON.stringify(res.productCreate.userErrors)}`);
  }

  return res.productCreate.product;
}

Module C: Margin Guard & Meta Ad Pause Engine (marginGuard.ts)

Calculates real-time unit margins and triggers an automated campaign freeze if an ad set breaches safety thresholds.

Net Profit = Attributed Revenue - Total COGS - Current Ad Spend - (Attributed Revenue × 0.029 + $0.30) 
import axios from 'axios';

const META_API_VERSION = 'v26.0';

export async function getHourlyAdSpend(adSetId: string, accessToken: string): Promise<number> {
  const response = await axios.get(
    `https://graph.facebook.com/${META_API_VERSION}/${adSetId}/insights`,
    {
      params: {
        access_token: accessToken,
        fields: 'spend',
        date_preset: 'today'
      }
    }
  );

  const data = response.data.data;
  return data && data.length > 0 ? parseFloat(data[0].spend) : 0.00;
}

export async function pauseAdSet(adSetId: string, accessToken: string): Promise<boolean> {
  const response = await axios.post(
    `https://graph.facebook.com/${META_API_VERSION}/${adSetId}`,
    { status: 'PAUSED' },
    { params: { access_token: accessToken } }
  );

  return response.data.success === true;
}

export async function runSafetyCircuitCheck(
  adSetId: string,
  metaToken: string,
  attributedRevenue: number,
  totalCogs: number,
  spendThreshold: number = 50.00
): Promise<{ actionTaken: string; netProfit: number }> {
  const currentSpend = await getHourlyAdSpend(adSetId, metaToken);
  const processorFees = attributedRevenue > 0 ? (attributedRevenue * 0.029) + 0.30 : 0;
  const netProfit = attributedRevenue - totalCogs - currentSpend - processorFees;

  if (currentSpend >= spendThreshold && netProfit < 0) {
    await pauseAdSet(adSetId, metaToken);
    return { actionTaken: 'KILLED_CAMPAIGN', netProfit };
  }

  return { actionTaken: 'MAINTAINED', netProfit };
}

5. Implementation Roadmap

  • Phase 1: Ingestion & Scraping Microservice (Days 1–10): Deploy Playwright worker pool with residential proxy rotation; parse hydration objects from AliExpress/CJ.
  • Phase 2: Content Transformation & Shopify Storefront (Days 11–18): Image cleaning pipeline via Sharp; LLM copy structuring; Shopify GraphQL Admin API product pipeline.
  • Phase 3: Real-Time Margin Guard & Webhooks (Days 19–28): Shopify orders/create webhooks; Meta v26.0 and TikTok marketing APIs; 15-minute cron circuit breaker.
  • Phase 4: Dashboard UI & Batch Fulfillment Session (Days 29–35): Real-time net margin dashboard; one-click batch supplier checkout and address injection.
